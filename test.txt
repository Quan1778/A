local function getImageAsset(cfg)
	if not cfg then return "" end
	if getcustomasset and isfile then
		if makefolder and cfg.LocalFileName and cfg.LocalFileName ~= "" and string.find(cfg.LocalFileName, "/") then
			local folderName = string.match(cfg.LocalFileName, "^(.*)/")
			if folderName and isfolder and not isfolder(folderName) then
				pcall(function() makefolder(folderName) end)
			end
		end

		if cfg.LocalFileName and cfg.LocalFileName ~= "" then
			local exists = false
			pcall(function() exists = isfile(cfg.LocalFileName) end)
			if exists then
				local success, asset = pcall(function() return getcustomasset(cfg.LocalFileName) end)
				if success and asset then return asset end
			end
		end
		
		if writefile and game.HttpGet and cfg.WebUrl and cfg.WebUrl ~= "" and cfg.LocalFileName and cfg.LocalFileName ~= "" then
			local success, bytes = pcall(function() return game:HttpGet(cfg.WebUrl) end)
			if success and bytes and #bytes > 0 then
				pcall(function() writefile(cfg.LocalFileName, bytes) end)
				local exists = false
				pcall(function() exists = isfile(cfg.LocalFileName) end)
				if exists then
					local successAsset, asset = pcall(function() return getcustomasset(cfg.LocalFileName) end)
					if successAsset and asset then return asset end
				end
			end
		end
	end
	return (cfg.FallbackAsset and cfg.FallbackAsset ~= "") and cfg.FallbackAsset or ""
end

local function toggleFly(state)
	flyEnabled = state
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	if flyEnabled then
		local bv = Instance.new("BodyVelocity")
		bv.Name = "AquaFlyBV"
		bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
		bv.Velocity = Vector3.zero
		bv.Parent = hrp

		local bg = Instance.new("BodyGyro")
		bg.Name = "AquaFlyBG"
		bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
		bg.CFrame = hrp.CFrame
		bg.Parent = hrp

		flyConnection = RunService.RenderStepped:Connect(function()
			if not flyEnabled or not hrp or not hrp.Parent then
				if flyConnection then flyConnection:Disconnect(); flyConnection = nil end
				return
			end
			local cam = workspace.CurrentCamera
			local moveDir = Vector3.zero
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + cam.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - cam.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - cam.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + cam.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

			bv.Velocity = moveDir * flySpeed
			bg.CFrame = cam.CFrame
		end)
	else
		if flyConnection then flyConnection:Disconnect(); flyConnection = nil end
		if hrp:FindFirstChild("AquaFlyBV") then hrp.AquaFlyBV:Destroy() end
		if hrp:FindFirstChild("AquaFlyBG") then hrp.AquaFlyBG:Destroy() end
	end
end

local function toggleAntiAfk(state)
	antiAfkEnabled = state
	if antiAfkEnabled then
		local VirtualUser = game:GetService("VirtualUser")
		antiAfkConn = LocalPlayer.Idled:Connect(function()
			VirtualUser:CaptureController()
			VirtualUser:ClickButton2(Vector2.zero)
		end)
	else
		if antiAfkConn then
			antiAfkConn:Disconnect()
			antiAfkConn = nil
		end
	end
end

	freezeEnabled = state

	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp then
		hrp.Anchored = freezeEnabled
	end
end

local function button(parent, text, size, pos)
	local b = Instance.new("TextButton", parent)
	b.Size = size; b.Position = pos; b.BackgroundColor3 = C.panel2; b.BorderSizePixel = 0
	b.Text = text; b.TextColor3 = C.text; b.Font = Enum.Font.GothamBold; b.TextSize = 11
	b.AutoButtonColor = false; b.Active = true; b.Selectable = true
	corner(b, 10)

	b.MouseEnter:Connect(function()
		if UserInputService.TouchEnabled then return end
		TweenService:Create(b, TweenInfo.new(0.15), {BackgroundColor3 = C.hover}):Play()
	end)
	b.MouseLeave:Connect(function()
		if UserInputService.TouchEnabled then return end
		TweenService:Create(b, TweenInfo.new(0.15), {BackgroundColor3 = C.panel2}):Play()
	end)

	b.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			TweenService:Create(b, TweenInfo.new(0.08), {
				BackgroundColor3 = C.dark,
				Size = UDim2.new(size.X.Scale, size.X.Offset - 2, size.Y.Scale, size.Y.Offset - 2)
			}):Play()
		end
	end)
	b.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			TweenService:Create(b, TweenInfo.new(0.12), {
				BackgroundColor3 = C.panel2,
				Size = size
			}):Play()
		end
	end)

	return b
end

		if name == tabName then
			data.Button.BackgroundColor3 = C.panel2
			data.Button.TextColor3 = C.aqua
			data.Indicator.Visible = true

	PageMain.Visible = (tabName == "MAIN")
	PageTeleport.Visible = (tabName == "TELEPORT")
	PageESP.Visible = (tabName == "ESP")
end

local frameCount = 0

RunService.RenderStepped:Connect(function()
	frameCount = frameCount + 1
	local now = os.clock()
	if now - lastFpsUpdate >= 0.5 then
		local fps = math.floor((frameCount / (now - lastFpsUpdate)) * 100) / 100
		FpsValue.Text = string.format("%.2f", fps)
		if fps >= 50 then
			FpsValue.TextColor3 = C.green
		elseif fps >= 30 then
			FpsValue.TextColor3 = Color3.fromRGB(240, 180, 50)
		else
			FpsValue.TextColor3 = C.red
		end
		frameCount = 0
		lastFpsUpdate = now
	end

	local ping = 0
	pcall(function()
		ping = StatsService.Network.ServerStatsItem["Data Ping"]:GetValue()
	end)
	PingValue.Text = string.format("%.2fms", ping)
	if ping < 100 then
		PingValue.TextColor3 = C.green
	elseif ping < 200 then
		PingValue.TextColor3 = Color3.fromRGB(240, 180, 50)
	else
		PingValue.TextColor3 = C.red
	end

	local sTime = math.floor(workspace.DistributedGameTime)
	local hrs = math.floor(sTime / 3600)
	local mins = math.floor((sTime % 3600) / 60)
	local secs = sTime % 60
	ServerTimeLbl.Text = string.format("Server Uptime: %02d:%02d:%02d", hrs, mins, secs)

	ServerInfoList.Text = string.format(
		"• Place ID: %d\n• Job ID: %s\n• Total Players: %d/%d\n• Executor: %s",
		game.PlaceId,
		string.sub(game.JobId, 1, 18) .. "...",
		#Players:GetPlayers(),
		Players.MaxPlayers,
		(identifyexecutor and identifyexecutor() or "Unknown")
	)
end)

local function animateToggle(knob, track, state)

	local targetTrackColor = state and C.green or C.panel2
	
	TweenService:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
		Position = targetKnobPos
	}):Play()

		BackgroundColor3 = targetTrackColor
	}):Play()
end

local function updateListHighlights()
	local selectedUserId = selected and selected.UserId or nil
	for _, item in ipairs(List:GetChildren()) do
		if item:IsA("TextButton") then
			local itemUserId = item:GetAttribute("TargetUserId")
			local isSelected = (itemUserId ~= nil and itemUserId == selectedUserId)
			
			local btnStroke = item:FindFirstChildOfClass("UIStroke")
			if isSelected then
				item.BackgroundColor3 = C.dark
				if btnStroke then
					btnStroke.Thickness = 2
					btnStroke.Transparency = 0
					btnStroke.Color = C.aqua
				end
			else
				item.BackgroundColor3 = C.panel2
				if btnStroke then
					btnStroke.Thickness = 0
					btnStroke.Transparency = 1
				end
			end
		end
	end
end

	updateListHighlights()
end

local function updatePreviewPosition(mousePos)
	local s = Scale.Scale or 1
	local posX = (mousePos.X / s) + 15
	local posY = (mousePos.Y / s) + 15
	
	local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
	if posX + PreviewFrame.Size.X.Offset > (viewport.X / s) then
		posX = (mousePos.X / s) - PreviewFrame.Size.X.Offset - 15
	end
	if posY + PreviewFrame.Size.Y.Offset > (viewport.Y / s) then
		posY = (mousePos.Y / s) - PreviewFrame.Size.Y.Offset - 15
	end

	PreviewFrame.Position = UDim2.fromOffset(posX, posY)
end

local function showPlayerPreview(plr, btn)
	selectPlayerDirectly(plr)
	currentHoveredPlayer = plr

	SkinImage.Image = "rbxthumb://type=Avatar&id=" .. plr.UserId .. "&w=420&h=420"
	InfoName.Text = plr.DisplayName
	InfoUser.Text = "@" .. plr.Name
	
	local r = root(plr.Character)
	if r then
		local pos = r.Position
		InfoPos.Text = string.format("X:%.1f Y:%.1f Z:%.1f", pos.X, pos.Y, pos.Z)
		WorldHighlight.Adornee = plr.Character
		WorldHighlight.Enabled = true
	else
		InfoPos.Text = "Pos: Out of range"
		WorldHighlight.Enabled = false
		notify("Target character out of render range", false)
	end
	
	PreviewFrame.Visible = true
end

local function getPlayerBtnAtPosition(pos2D)
	for _, item in ipairs(List:GetChildren()) do
		if item:IsA("TextButton") and item:GetAttribute("TargetUserId") then
			local absPos = item.AbsolutePosition
			local absSize = item.AbsoluteSize
			if pos2D.X >= absPos.X and pos2D.X <= absPos.X + absSize.X and
			   pos2D.Y >= absPos.Y and pos2D.Y <= absPos.Y + absSize.Y then
				local uid = item:GetAttribute("TargetUserId")
				for _, p in ipairs(Players:GetPlayers()) do
					if p.UserId == uid then
						return item, p
					end
				end
			end
		end
	end
	return nil, nil
end

local function refresh()
	clear3DHighlight()
	
	for _,v in ipairs(List:GetChildren()) do
		if v:IsA("TextButton") then v:Destroy() end
	end
	
	local filter = string.lower(SearchBox.Text)

	for _,plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer then
			local dName = string.lower(plr.DisplayName)
			local uName = string.lower(plr.Name)
			
			if filter == "" or string.find(dName, filter, 1, true) or string.find(uName, filter, 1, true) then
				local isSel = selected and plr.UserId == selected.UserId
				local b = Instance.new("TextButton", List)
				b.Size = UDim2.new(1, -2, 0, 42)
				b.BackgroundColor3 = isSel and C.dark or C.panel2
				b.BorderSizePixel = 0
				b.Text = ""
				b.AutoButtonColor = false
				b.Active = true
				b:SetAttribute("TargetUserId", plr.UserId)
				corner(b, 8)

				local btnStroke = Instance.new("UIStroke", b)
				btnStroke.Color = C.aqua
				btnStroke.Thickness = isSel and 2 or 0
				btnStroke.Transparency = isSel and 0 or 1
				
				local avatar = Instance.new("ImageLabel", b)
				avatar.Size = UDim2.fromOffset(30, 30)
				avatar.Position = UDim2.fromOffset(6, 6)
				avatar.BackgroundTransparency = 1
				avatar.Image = "rbxthumb://type=AvatarHeadShot&id=" .. plr.UserId .. "&w=150&h=150"
				corner(avatar, 15)
				
				local nameLbl = Instance.new("TextLabel", b)
				nameLbl.Size = UDim2.new(1, -48, 1, 0)
				nameLbl.Position = UDim2.fromOffset(44, 0)
				nameLbl.BackgroundTransparency = 1
				nameLbl.Text = plr.DisplayName
				nameLbl.TextColor3 = C.text
				nameLbl.Font = Enum.Font.GothamBold
				nameLbl.TextSize = 11
				nameLbl.TextXAlignment = Enum.TextXAlignment.Left

				b.MouseEnter:Connect(function()
					if UserInputService.TouchEnabled then return end
					if not selected or selected.UserId ~= plr.UserId then
						TweenService:Create(b, TweenInfo.new(0.15), {BackgroundColor3 = C.hover}):Play()
					end
				end)
				b.MouseLeave:Connect(function()
					if UserInputService.TouchEnabled then return end
					if not selected or selected.UserId ~= plr.UserId then
						TweenService:Create(b, TweenInfo.new(0.15), {BackgroundColor3 = C.panel2}):Play()
					end
				end)

				b.Activated:Connect(function()
					cancelHoldTimer()
					if not activeHolding then
						selectPlayerDirectly(plr)
					end
				end)

				b.InputBegan:Connect(function(input)
					if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
						cancelHoldTimer()
						holdDragInput = input
						startTouchPos = input.Position
						pendingPlayer = plr
						isScrolling = false

						holdTimerThread = task.delay(0.25, function()
							if not isScrolling then
								activeHolding = true
								updatePreviewPosition(input.Position)
								showPlayerPreview(pendingPlayer, b)
							end
						end)
					end
				end)
			end
		end
	end
	
	List.CanvasSize = UDim2.new(0, 0, 0, 0)
end

UserInputService.InputChanged:Connect(function(input)
	if holdDragInput and input == holdDragInput then
		if startTouchPos then
			local dist = (input.Position - startTouchPos).Magnitude
			if dist > 12 then
				isScrolling = true
				cancelHoldTimer()
				if activeHolding then
					activeHolding = false
					clear3DHighlight()
				end
			end
		end

		if activeHolding then
			updatePreviewPosition(input.Position)
			local btn, plrTarget = getPlayerBtnAtPosition(input.Position)
			if plrTarget and btn then
				showPlayerPreview(plrTarget, btn)
			end
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if holdDragInput and input == holdDragInput then
		cancelHoldTimer()
		if activeHolding then
			activeHolding = false
			clear3DHighlight()
		end
		startTouchPos = nil
		pendingPlayer = nil
		holdDragInput = nil
	end
end)

local function executeRandomTP()
	local availablePlayers = {}
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and p.Character and root(p.Character) then
			table.insert(availablePlayers, p)
		end
	end
	
	if #availablePlayers == 0 then
		notify("No available players found", false)
		return
	end
	
	local target = availablePlayers[math.random(1, #availablePlayers)]
	local r = root(LocalPlayer.Character)
	local tr = root(target.Character)
	
	if r and tr then
		savedCF = r.CFrame
		savedPos = r.Position
		r.CFrame = tr.CFrame + Vector3.new(0, 3, 0)
		selectPlayerDirectly(target)
		notify("Random TP to " .. target.DisplayName, true)
	else
		notify("Target lost", false)
	end
end

local function applyESPToPlayer(plr)
	if plr == LocalPlayer then return end

	local function setupChar(char)
		if not char then return end
		local head = char:WaitForChild("Head", 3)
		if not head then return end

		local hl = char:FindFirstChild("AquaESP_HL") or Instance.new("Highlight")
		hl.Name = "AquaESP_HL"
		hl.FillColor = C.aqua
		hl.OutlineColor = Color3.fromRGB(255, 255, 255)
		hl.FillTransparency = 0.5
		hl.OutlineTransparency = 0
		hl.Parent = char

		local bb = head:FindFirstChild("AquaESP_Tag") or Instance.new("BillboardGui")
		bb.Name = "AquaESP_Tag"
		bb.Size = UDim2.fromOffset(150, 30)
		bb.StudsOffset = Vector3.new(0, 2.5, 0)
		bb.AlwaysOnTop = true
		bb.Parent = head

		local tagLbl = bb:FindFirstChild("TagLbl") or Instance.new("TextLabel", bb)
		tagLbl.Name = "TagLbl"
		tagLbl.Size = UDim2.fromScale(1, 1)
		tagLbl.BackgroundTransparency = 1
		tagLbl.TextColor3 = C.text
		tagLbl.Font = Enum.Font.GothamBold
		tagLbl.TextSize = 11
		tagLbl.TextStrokeTransparency = 0.2
	end

	plr.CharacterAdded:Connect(setupChar)
	if plr.Character then setupChar(plr.Character) end
end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character then
			local char = plr.Character

							tagLbl.Text = string.format("%s\n[%d m]", plr.DisplayName, dist)

					end
				end
			end
		end
	end

Bubble.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		bubbleDragInput = input
		hasDraggedBubble = false
		bubbleDragStart = input.Position
		bubbleStartPos = Bubble.Position
		
		local conn; conn = input:GetPropertyChangedSignal("UserInputState"):Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				bubbleDragInput = nil
				conn:Disconnect()
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if bubbleDragInput and input == bubbleDragInput then
		local delta = input.Position - bubbleDragStart
		if delta.Magnitude > 6 then
			hasDraggedBubble = true
		end
		
		local scaleFactor = Scale.Scale or 1
		local targetPos = UDim2.new(
			bubbleStartPos.X.Scale,
			bubbleStartPos.X.Offset + (delta.X / scaleFactor),
			bubbleStartPos.Y.Scale,
			bubbleStartPos.Y.Offset + (delta.Y / scaleFactor)
		)
		
		TweenService:Create(Bubble, TweenInfo.new(0.06, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Position = targetPos
		}):Play()
	end
end)

	if not hasDraggedBubble then
		Bubble.Visible = false
		Main.Visible = true
	end
end)
